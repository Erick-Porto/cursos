## Visão do encontro

**O que observar:** `itens.push(...)` na lista recebida; `item.qtd++` ou `item.qtd = qtd` dentro da regra, que muda o item de quem chamou; `console.log` em `src/regras/`; `alterarQuantidade` sem o caso `qtd <= 0`; `adicionarItem` que duplica a linha do produto repetido; acumulador criado dentro do laço; import relativo sem `.js`; regra chamada no script sem guardar o resultado. Para quem travou: "o que entra nesta função e o que sai dela?".

:::example Solução de referência
`src/regras/carrinho.js`, primeira parte (a `subtotal` é a do Bloco 1):

```javascript
import { paraCentavos, paraReais } from './precos.js';

// Regras puras do carrinho: recebem dados e devolvem dados, sem imprimir nada.

export function subtotal(itens) {
  let centavos = 0;
  for (const item of itens) {
    centavos += paraCentavos(item.preco) * item.qtd;
  }
  return paraReais(centavos);
}

// Uma linha do carrinho: só o que o carrinho precisa do produto, mais a quantidade.
function criarItem(produto, qtd) {
  return { id: produto.id, nome: produto.nome, preco: produto.preco, qtd: qtd };
}

export function adicionarItem(itens, produto) {
  const novos = [];
  let achou = false;
  for (const item of itens) {
    if (item.id === produto.id) {
      novos.push(criarItem(item, item.qtd + 1));
      achou = true;
    } else {
      novos.push(item);
    }
  }
  if (!achou) {
    novos.push(criarItem(produto, 1));
  }
  return novos;
}
```

`src/regras/carrinho.js`, segunda parte:

```javascript
export function removerItem(itens, id) {
  const novos = [];
  for (const item of itens) {
    if (item.id !== id) {
      novos.push(item);
    }
  }
  return novos;
}

export function alterarQuantidade(itens, id, qtd) {
  if (qtd <= 0) {
    return removerItem(itens, id);
  }
  const novos = [];
  for (const item of itens) {
    if (item.id === id) {
      novos.push(criarItem(item, qtd));
    } else {
      novos.push(item);
    }
  }
  return novos;
}
```

`src/regras/carrinho.test.js`, primeira parte:

```javascript
import { adicionarItem, alterarQuantidade, removerItem, subtotal } from './carrinho.js';

const fone = { id: 1, nome: 'Fone Bluetooth', preco: 199.9 };
const luminaria = { id: 4, nome: 'Luminária de Mesa', preco: 89.9 };

describe('subtotal', () => {
  test('soma preço vezes quantidade sem sobra de centavos', () => {
    const itens = [
      { id: 1, nome: 'Fone Bluetooth', preco: 199.9, qtd: 2 },
      { id: 4, nome: 'Luminária de Mesa', preco: 89.9, qtd: 1 },
    ];
    expect(subtotal(itens)).toBe(489.7);
  });

  test('carrinho vazio vale zero', () => {
    expect(subtotal([])).toBe(0);
  });
});

describe('adicionarItem', () => {
  test('produto novo entra com qtd 1', () => {
    expect(adicionarItem([], fone)).toEqual([{ id: 1, nome: 'Fone Bluetooth', preco: 199.9, qtd: 1 }]);
  });

  test('produto repetido soma 1 na qtd, sem duplicar a linha', () => {
    const novos = adicionarItem([{ id: 1, nome: 'Fone Bluetooth', preco: 199.9, qtd: 1 }], fone);
    expect(novos).toHaveLength(1);
    expect(novos[0].qtd).toBe(2);
  });

  test('não altera o carrinho recebido', () => {
    const itens = [{ id: 1, nome: 'Fone Bluetooth', preco: 199.9, qtd: 1 }];
    adicionarItem(itens, fone);
    adicionarItem(itens, luminaria);
    expect(itens).toHaveLength(1);
    expect(itens[0].qtd).toBe(1);
  });
});
```

describe('removerItem', () => {
  test('tira só o item do id informado', () => {
    const itens = [
      { id: 1, nome: 'Fone Bluetooth', preco: 199.9, qtd: 1 },
      { id: 4, nome: 'Luminária de Mesa', preco: 89.9, qtd: 2 },

  test('id inexistente devolve os mesmos itens', () => {
    const itens = [{ id: 1, nome: 'Fone Bluetooth', preco: 199.9, qtd: 1 }];
    expect(removerItem(itens, 99)).toEqual(itens);
  });
});

describe('alterarQuantidade', () => {
  test('troca a qtd só do item informado', () => {
    const itens = [
      { id: 1, nome: 'Fone Bluetooth', preco: 199.9, qtd: 1 },
      { id: 4, nome: 'Luminária de Mesa', preco: 89.9, qtd: 2 },

    expect(novos[1].qtd).toBe(5);

import { produtos } from '../data/produtos.js';
import { adicionarItem, alterarQuantidade, removerItem, subtotal } from '../regras/carrinho.js';

let itens = [];
itens = adicionarItem(itens, produtos[0]); // Fone Bluetooth

itens = alterarQuantidade(itens, 4, 3);

for (const item of itens) {
  console.log(`${item.qtd} x ${item.nome}: ${formatarPreco(item.preco * item.qtd)}`);

console.log(`Subtotal: ${formatarPreco(subtotal(itens))}`);

## Dificuldades frequentes

| Sintoma | Causa provável | Como ajudar |
|---|---|---|
| `Received: 489.70000000000005` | Soma em reais | `paraCentavos` na soma e `paraReais` no fim |
| `Received: NaN` no subtotal | `for...in` no lugar de `for...of` | Trocar `in` por `of`; um `console.log(item)` dentro do laço mostra `'0'` |
| `Cannot read properties of undefined (reading 'preco')` | `i <= itens.length` no `for` clássico | Teste de mesa com 2 itens no papel |
| `Cannot find module '...\src\regras\precos'` ao rodar o script | Import relativo sem `.js` no fim | `from './precos.js'`: com módulos ES no Node, a extensão é obrigatória |
| Teste "não altera o carrinho recebido" vermelho | `itens.push(...)` ou `item.qtd = ...` na regra | Lista nova com `const novos = []` e objeto novo (`criarItem`) para o item alterado |
| Produto repetido aparece em duas linhas | `adicionarItem` sem conferir se o id já está no carrinho | Variável `achou` antes do laço, `if` dentro dele |
| `alterarQuantidade(itens, 1, 0)` deixa `qtd: 0` | Falta o caso `qtd <= 0` | `if (qtd <= 0) { return removerItem(itens, id); }` no começo |
| O script imprime o carrinho vazio | Regra chamada sem guardar o resultado | `let itens = [];` e `itens = adicionarItem(itens, produto);` |
| `Cannot access '...' before initialization` | Arrow chamada antes da linha que a cria | Mover a chamada ou usar declaração |
