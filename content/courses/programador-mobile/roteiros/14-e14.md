`src/regras/frete.js`, o que entra (o `import` vai para o topo do arquivo):

```javascript
import { paraCentavos } from './precos.js';

export function unidadesParaFreteGratis(subtotal, preco) {
  if (paraCentavos(preco) <= 0) return null; // sem preço, o laço nunca chegaria à meta
  const meta = paraCentavos(FRETE_GRATIS_A_PARTIR_DE);
  let centavos = paraCentavos(subtotal);
  let unidades = 0;
  while (centavos < meta) {
    centavos += paraCentavos(preco);
    unidades++;
  }
  return unidades;
}
```

`src/regras/catalogo.js`, o que entra abaixo do `aplicarFiltros` do E08, primeira parte:

```javascript
function normalizar(texto) {
  return texto.normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase();
}

function contemTodas(texto, palavras) {
  for (const palavra of palavras) {
    if (!texto.includes(palavra)) return false; // uma faltou: nem olha as outras
  }
  return true;
}

export function buscarProdutos(produtos, termo) {
  const palavras = normalizar(termo ?? '').split(' ').filter((p) => p !== '');
  if (palavras.length === 0) return produtos; // termo vazio: a vitrine toda
  const encontrados = [];
  for (const produto of produtos) {
    if (contemTodas(normalizar(produto.nome), palavras)) {
      encontrados.push(produto);
    }
  }
  return encontrados;
}
```

`src/regras/catalogo.js`, segunda parte, com o `paginar` já trocado para `slice` no passo 3:

```javascript
export function itemMaisCaro(itens) {
  let maisCaro = null;
  for (const item of itens) {
    if (maisCaro === null || item.preco > maisCaro.preco) {
      maisCaro = item;
    }
  }
  return maisCaro;
}

export function paginar(lista, pagina = 1, tamanho = 6) {
  if (pagina < 1 || tamanho < 1) return [];
  const inicio = (pagina - 1) * tamanho;
  return lista.slice(inicio, inicio + tamanho); // o slice já para no fim da lista
}
```

`src/scripts/buscar.js`:

```javascript
import { produtos } from '../data/produtos.js';
import { buscarProdutos, paginar } from '../regras/catalogo.js';
import { formatarPreco } from '../utils/formatar.js';

const args = process.argv.slice(2);
const opcao = args.find((a) => a.startsWith('--pagina='));
const pagina = opcao ? Number(opcao.split('=')[1]) : 1;
const termo = args.filter((a) => !a.startsWith('--')).join(' ');

const encontrados = buscarProdutos(produtos, termo);
console.log(`${encontrados.length} produto(s) para "${termo}", página ${pagina}`);
for (const produto of paginar(encontrados, pagina, 5)) {
  console.log(`  ${produto.nome}: ${formatarPreco(produto.preco)}`);
}
```

## Dificuldades frequentes

| Sintoma | Causa provável | Como ajudar |
|---|---|---|
| `NaN` no total, ou "R$ NaN" no relatório | `for...in` no lugar de `for...of` | Trocar `in` por `of`; mostrar `typeof item` no breakpoint |
| `TypeError: porCategoria is not iterable` | `for...of` sobre um objeto comum | `for...in` ou `Object.entries` (E08) |
| `npm test` ou o script não terminam | Laço infinito: preço 0, `i--` ou `continue` antes do passo | <kbd>Ctrl</kbd> + <kbd>C</kbd>; conferir condição e passo; guard clause |
| A busca não acha "tenis" | Normalizou só um lado | Rodar o teste do "TENIS"; conferir as duas chamadas de `normalizar` |
| "fone mesa" acha produtos demais | `return true` dentro do laço de `contemTodas` | Teste de mesa com 2 palavras: o `return true` vem depois do laço |
| `itemMaisCaro` devolve o último no empate | `>=` no lugar de `>` | Rodar o teste do empate e ler a condição em voz alta |
| `Cannot find module '.../catalogo'` | Import relativo sem `.js` | Módulos ES pedem a extensão: `'./catalogo.js'` |
| O breakpoint não pausa | Script rodado num terminal comum | Abrir o JavaScript Debug Terminal e rodar de novo |
