`src/regras/carrinho.js`, completo (o `criarItem` do E07 continua igual):

```javascript
import { paraCentavos, paraReais } from './precos.js';

// Regras puras do carrinho: recebem dados e devolvem dados, sem imprimir nada.

export function subtotal(itens) {
  const centavos = itens.reduce(
    (soma, item) => soma + paraCentavos(item.preco) * item.qtd,
    0,
  );
  return paraReais(centavos);
}

export function contarItens(itens) {
  return itens.reduce((total, item) => total + item.qtd, 0);
}

// Uma linha do carrinho: só o que o carrinho precisa do produto, mais a quantidade.
function criarItem(produto, qtd) {
  return { id: produto.id, nome: produto.nome, preco: produto.preco, qtd: qtd };
}

export function adicionarItem(itens, produto) {
  const existente = itens.find((item) => item.id === produto.id);
  if (existente !== undefined) {
    return alterarQuantidade(itens, produto.id, existente.qtd + 1);
  }
  return [...itens, criarItem(produto, 1)];
}

export function removerItem(itens, id) {
  return itens.filter((item) => item.id !== id);
}

export function alterarQuantidade(itens, id, qtd) {
  if (qtd <= 0) {
    return removerItem(itens, id);
  }
  return itens.map((item) => {
    if (item.id === id) {
      return { ...item, qtd };
    }
    return item;
  });
}
```

As três regras passam de cerca de 40 linhas para 21. `adicionarItem` chama `alterarQuantidade`, declarada mais abaixo: funções declaradas com `function` podem ser usadas antes da linha em que aparecem.

```javascript
export function aplicarFiltros(lista, { categoria, ordem = 'nome' } = {}) {
  let base = lista;
  if (categoria) {
    base = lista.filter((produto) => produto.categoria === categoria);
  }
  const comparar = comparadores[ordem] || comparadores.nome;
  return [...base].sort(comparar);
}

export function agruparPorCategoria(produtos) {
  return produtos.reduce((grupos, produto) => {
    const slug = produto.categoria;
    if (grupos[slug] === undefined) {
      grupos[slug] = [];
    }
    grupos[slug].push(produto);
    return grupos;
  }, {});
}
```

O `|| comparadores.nome` continua: o padrão `'nome'` cobre a ordem que não veio, e o `||` cobre a ordem desconhecida, como `'cor'`. O `push` do agrupamento é seguro: ele mexe nos arrays que a própria função criou, não na lista recebida. O critério "nenhum `push`" vale só para `carrinho.js`.

## Dificuldades frequentes

| Sintoma | Causa provável | Como ajudar |
|---|---|---|
| `[object Object]` no resultado ou `Reduce of empty array with no initial value` | `reduce` sem valor inicial | Perguntar quanto vale o acumulador na primeira volta; acrescentar `0` ou `{}` |
| `map` devolve `[undefined, undefined]` | Arrow com chaves e sem `return` | Tire as chaves ou escreva o `return`; com objeto, use parênteses |
| Teste do subtotal falha com `379.70000000000005` ou parecido | Soma em reais, sem `paraCentavos` | Somar em centavos e voltar com `paraReais` no fim |
| Teste "não altera o carrinho recebido" do E07 vermelho | `push`, `existente.qtd += 1` ou `item.qtd = qtd` no carrinho recebido, inclusive depois de `[...itens]` | Trocar pela coluna da direita de "Atualizar sem mutar"; no item alterado, `{ ...item, qtd }` |
| Produto repetido aparece em duas linhas | `find` comparando `item.id === produto` em vez de `produto.id`, ou `if (existente === null)` | O `find` devolve `undefined` quando não acha; comparar os dois `id` |
| `console.log(itens)` mostra `categoria`, `imagem` e `descricao` no carrinho | `{ ...produto, qtd: 1 }` no lugar de `criarItem(produto, 1)` | O item do carrinho guarda só `{ id, nome, preco, qtd }`: voltar ao `criarItem` |
| `Cannot destructure property 'categoria' of 'undefined'` | Parâmetro desestruturado sem `= {}`, e `aplicarFiltros(produtos)` chamado sem filtros | Acrescentar `= {}` depois das chaves do parâmetro |
| `Cannot read properties of undefined (reading 'push')` | O array da categoria não foi criado antes do `push` | `if (grupos[slug] === undefined) { grupos[slug] = []; }` antes do `push` |
| `Cannot read properties of undefined` na segunda volta do `reduce` | Falta `return grupos` no fim da arrow com chaves | Cada volta precisa devolver o acumulador |
| `categorias.js` com `Cannot read properties of undefined (reading 'length')` | Leu `grupos[categoria.nome]` em vez de `grupos[categoria.slug]` | `console.log(Object.keys(grupos))` e comparar com os `slug` |
| `ERR_MODULE_NOT_FOUND` ao rodar um script ou os testes | Import relativo sem `.js`, ou `./` no lugar de `../` | Conferir a extensão e a pasta: os scripts estão em `src/scripts/` |
| Projeto 2: `contarPorGrupo` dá `NaN` ou esquece o grupo vazio | Valor inicial `{}` no `reduce` | Valor inicial com os três grupos em zero |
