`src/utils/filtrar.js`:

```javascript
const porNome = (a, b) => a.nome.localeCompare(b.nome, 'pt-BR');

export const comparadores = {
  nome: porNome,
  preco: (a, b) => a.preco - b.preco || porNome(a, b),
  avaliacao: (a, b) => b.avaliacao - a.avaliacao || porNome(a, b),
};

export const rotulosDeOrdem = {
  nome: 'Nome',
  preco: 'Menor preço',
  avaliacao: 'Melhor avaliação',
};

export function aplicarFiltros(lista, { categoria = null, ordem = 'nome' } = {}) {
  const base = categoria ? lista.filter((p) => p.categoria === categoria) : lista;
  const comparar = comparadores[ordem] ?? comparadores.nome;
  return [...base].sort(comparar);
}
```

`src/regras/historico.js`:

```javascript
export function registrarBusca(historico, termo, limite = 5) {
  const limpo = termo.trim();
  if (limpo === '') return historico;

  const novo = [...historico];
  const repetida = novo.findIndex((t) => t.toLowerCase() === limpo.toLowerCase());
  if (repetida !== -1) novo.splice(repetida, 1);

  novo.unshift(limpo);
  return novo.slice(0, limite);
}
```

`src/regras/favoritos.js`:

```javascript
export function ehFavorito(favoritos, id) {
  return favoritos.includes(id);
}

export function alternarFavorito(favoritos, id) {
  const novo = [...favoritos];
  const posicao = novo.indexOf(id);
  if (posicao === -1) {
    novo.push(id);
  } else {
    novo.splice(posicao, 1);
  }
  return novo;
}
```

## Dificuldades frequentes

| Sintoma | Causa provável | Como ajudar |
|---|---|---|
| Categorias muda de ordem depois de ordenar a vitrine | `sort` direto em `base`, que sem categoria é o catálogo | `[...base].sort(...)` e o teste "não altera a lista recebida" |
| Uma ordem não tem efeito e a grade fica por nome | Chaves diferentes entre `rotulosDeOrdem` e `comparadores` | `Object.keys` dos dois no REPL; o teste das chaves |
| Teste com arrays iguais falha | `toBe` comparando dois arrays com o mesmo conteúdo | Trocar por `toEqual` |
| O histórico ganha uma entrada por letra | `registrarBusca` no `onChangeText` | Mover para `onSubmitEditing`, com `returnKeyType="search"` |
| A mesma busca aparece duas vezes | Falta o `splice` da repetida, ou o `findIndex` não usa `toLowerCase` | Conferir o `findIndex` e a comparação em minúsculas |
| Tela vermelha `favoritos.includes is not a function` | `setFavoritos(favoritos.push(id))` guarda o número que o `push` devolve | Usar `alternarFavorito`, que devolve o array novo |
| Tocar no coração abre o detalhe | Coração dentro do `Pressable` do cartão, ou antes dele no JSX | Coração depois do cartão, dentro da `View`, com `position: 'absolute'` |
| "Rendered more hooks than during the previous render" | `useState` novo depois de um `return` antecipado | Subir os hooks para o topo do componente |
