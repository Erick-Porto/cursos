```javascript
const inicio = Date.now();
const buscar = debounce((termo) => console.log(`${Date.now() - inicio} ms: buscando "${termo}"`), 300);
const INTERVALO = Number(process.argv[2] ?? 120); // ms entre uma letra e outra
['f', 'fo', 'fon', 'fone'].forEach((termo, i) => setTimeout(() => buscar(termo), i * INTERVALO));
```

:::example Solução de referência
`src/utils/funcoes.js`:

```javascript
export function memoize(fn) {
  const cache = {};
  return (...args) => {
    const chave = JSON.stringify(args);
    if (chave in cache) return cache[chave];
    const resultado = fn(...args);
    cache[chave] = resultado;
    return resultado;
  };
}

export function debounce(fn, espera = 300) {
  let temporizador;
  return (...args) => {
    clearTimeout(temporizador);
    temporizador = setTimeout(() => fn(...args), espera);
  };
}
```

`src/utils/funcoes.test.js`, primeira parte:

```javascript
import { debounce, memoize } from './funcoes';

jest.useFakeTimers();

describe('memoize', () => {
  test('roda a função original uma vez para os mesmos argumentos', () => {
    const calcular = jest.fn((subtotal, cep) => (cep.startsWith('0') ? 15 : 25));
    const calcularComCache = memoize(calcular);
    expect(calcularComCache(80, '01001000')).toBe(15);
    expect(calcularComCache(80, '01001000')).toBe(15);
    expect(calcular).toHaveBeenCalledTimes(1);
  });

  test('argumentos diferentes calculam de novo', () => {
    const calcular = jest.fn((subtotal, cep) => (cep.startsWith('0') ? 15 : 25));
    const calcularComCache = memoize(calcular);
    calcularComCache(80, '01001000');
    expect(calcularComCache(80, '90010000')).toBe(25);
    expect(calcular).toHaveBeenCalledTimes(2);
  });

  test('guarda também um resultado null', () => {
    const validar = jest.fn(() => null);
    const validarComCache = memoize(validar);
    validarComCache('123');
    expect(validarComCache('123')).toBeNull();
    expect(validar).toHaveBeenCalledTimes(1);
  });
});
```

```javascript
  test('busca uma vez, com o último termo, depois da pausa', () => {
```

```javascript
import { useEffect, useState } from 'react';

export function useDebounce(valor, espera = 300) {
  const [atrasado, setAtrasado] = useState(valor);

  useEffect(() => {
    const temporizador = setTimeout(() => setAtrasado(valor), espera);
    return () => clearTimeout(temporizador);
  }, [valor, espera]);

  return atrasado;
}
```

## Dificuldades frequentes

| Sintoma | Causa provável | Como ajudar |
|---|---|---|
| A busca continua filtrando a cada letra | A vitrine usa `termo` em vez de `termoAtrasado` | Trocar o argumento de `buscarProdutos` |
| As letras aparecem atrasadas no campo | `value={termoAtrasado}` no `Campo` | O campo fica com `termo`; só a lista usa o atrasado |
| A busca nunca atualiza depois da primeira letra | Dependências do efeito como `[]` | `[valor, espera]`: o efeito precisa rodar a cada valor novo |
| Teste do debounce: "Received number of calls: 0" | Sem `jest.useFakeTimers()` ou tempo avançado menor que a espera | Relógio falso no topo do arquivo; avançar 300 ms |
| `Element type is invalid ... got: undefined` | Import com chaves de um `export default` | `import Botao from '...'` ou o barril `../components/ui` |
| O teste "sem ponto" não lança | `cancelar = () => ...` como propriedade da classe | Escrever como método, `cancelar() { ... }`, igual ao `comStatus` |
