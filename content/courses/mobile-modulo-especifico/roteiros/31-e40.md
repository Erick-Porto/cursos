`src/hooks/useProdutos.js`, primeira parte:

```javascript
import { useEffect, useRef, useState } from 'react';
import { listarProdutos, pesquisar } from '../api/produtos';

const LIMITE = 20;

async function buscarVitrine(termo, pular, sinal) {
  if (!termo.trim()) return listarProdutos({ limite: LIMITE, pular, sinal });
  const encontrados = await pesquisar(termo, sinal); // a busca vem numa página só
  return { produtos: encontrados, total: encontrados.length };
}

export function useProdutos({ termo = '' } = {}) {
  const [produtos, setProdutos] = useState([]);
  const [total, setTotal] = useState(0);
  const [carregando, setCarregando] = useState(true);
  const [atualizando, setAtualizando] = useState(false);
  const [carregandoMais, setCarregandoMais] = useState(false);
  const [erro, setErro] = useState(null);
  const buscando = useRef(false);
  const sinal = useRef(null);

  async function buscarPagina(pular) {
    const meu = sinal.current;
    buscando.current = true;
    try {
      const pagina = await buscarVitrine(termo, pular, meu);
      setProdutos((atuais) => (pular === 0 ? pagina.produtos : [...atuais, ...pagina.produtos]));
      setTotal(pagina.total);
      setErro(null);
    } catch (e) {
      if (e.name !== 'AbortError') setErro(e.message);
    } finally {
      buscando.current = false;
    }
    return !meu.aborted;
  }
```

`src/hooks/useProdutos.js`, segunda parte:

```javascript
  async function recarregar() {
    setCarregando(true);
    const valeu = await buscarPagina(0);
    if (valeu) setCarregando(false);
  }

  useEffect(() => {
    const controle = new AbortController();
    sinal.current = controle.signal;
    recarregar();
    return () => controle.abort();
  }, [termo]);

  async function atualizar() {
    if (buscando.current) return;
    setAtualizando(true);
    await buscarPagina(0);
    setAtualizando(false);
  }

  async function carregarMais() {
    if (buscando.current || carregando || produtos.length >= total) return;
    setCarregandoMais(true);
    await buscarPagina(produtos.length);
    setCarregandoMais(false);
  }

  const temMais = produtos.length < total;
  return { produtos, carregando, atualizando, carregandoMais, erro, temMais,
    recarregar, atualizar, carregarMais };
}
```

## Dificuldades frequentes

| Sintoma | Causa provável | Como ajudar |
|---|---|---|
| Produtos repetidos e `Encountered two children with the same key` | Trava com `useState` ou sem trava | `useRef(false)`, ligado e desligado no `buscarPagina` |
| A grade fica sempre com 20, trocando a cada rolagem | `setProdutos(pagina.produtos)` em todas as páginas | Juntar com `[...atuais, ...pagina.produtos]` quando `pular` não é 0 |
| Os 20 primeiros produtos nunca aparecem | `skip` calculado com a página começando em 1 | `pular: produtos.length` |
| "Você viu todos os produtos" já na primeira página | `listarProdutos` ainda devolve só a lista: `total` fica `undefined` | Devolver `{ produtos, total }`; `console.log(pagina)` no hook |
| Tela vermelha no `aplicarFiltros` ao mudar o `listarProdutos` | A `buscarVitrine` antiga ficou no Início, entregando `{ produtos, total }` como lista | Apagar a do Início; a do hook já trata os dois formatos |
| A vitrine fica vazia com um filtro de categoria | O filtro ainda usa as categorias locais (audio, casa), que a DummyJSON não tem | Limpar o filtro hoje; a tarefa de casa leva Filtros à API |
| O esqueleto pisca ao puxar para atualizar | `atualizar` liga `carregando` | Ligar só `atualizando` |
| `npx expo lint` avisa `react-hooks/exhaustive-deps` no hook | O efeito usa `recarregar` sem listá-lo, de propósito | O efeito deve rodar só quando o termo muda: `// eslint-disable-next-line react-hooks/exhaustive-deps` |
