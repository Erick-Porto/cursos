:::example Solução de referência
`src/regras/favoritos.js`:

```javascript
export function ehFavorito(favoritos, id) {
  return favoritos.has(id);
}

export function alternarFavorito(favoritos, id) {
  const novo = new Set(favoritos); // o Set recebido não muda
  if (novo.has(id)) {
    novo.delete(id);
  } else {
    novo.add(id);
  }
  return novo;
}

export function listarFavoritos(favoritos, produtosPorId) {
  return [...favoritos]
    .map((id) => produtosPorId.get(id))
    .filter((produto) => produto !== undefined);
}
```

```javascript
import { alternarFavorito, ehFavorito, listarFavoritos } from './favoritos';

describe('ehFavorito', () => {
  test('responde pelo id, com o tipo certo', () => {
    expect(ehFavorito(new Set([3, 7]), 7)).toBe(true);
    expect(ehFavorito(new Set([3, 7]), '7')).toBe(false);
  });
});

describe('alternarFavorito', () => {
  test('liga um id que não estava', () => {
    expect(alternarFavorito(new Set([3]), 7)).toEqual(new Set([3, 7]));
  });

  test('desliga um id que já estava', () => {
    expect(alternarFavorito(new Set([3, 7]), 7)).toEqual(new Set([3]));
  });

  test('devolve um Set novo e não muda o recebido', () => {
    const original = new Set([3]);
    const novo = alternarFavorito(original, 7);
    expect(novo).not.toBe(original);
    expect(original).toEqual(new Set([3]));
  });
});
```

`src/contexts/FavoritosContext.js`:

```jsx
import { createContext, useContext, useState } from 'react';
import { alternarFavorito } from '../regras/favoritos';

const FavoritosContext = createContext(null);

export function FavoritosProvider({ children }) {
  const [favoritos, setFavoritos] = useState(new Set());

  function alternar(id) {
    setFavoritos((anterior) => alternarFavorito(anterior, id));
  }
  function ehFavorito(id) {
    return favoritos.has(id);
  }
  const valor = { favoritos, alternar, ehFavorito };
  return <FavoritosContext.Provider value={valor}>{children}</FavoritosContext.Provider>;
}

export function useFavoritos() {
  const valor = useContext(FavoritosContext);
  if (!valor) throw new Error('useFavoritos fora do FavoritosProvider');
  return valor;
}
```
