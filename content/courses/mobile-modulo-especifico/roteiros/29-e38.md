## Visão do encontro

**Objetivo:** trocar o `then` por `async`/`await` no checkout do E37 e conectar a Lojinha a uma API de verdade, com erros de rede e de resposta tratados, camada `src/api/`, endereço no `.env` e busca que cancela o pedido velho.

Commit "E38: async/await e catálogo da API".

**Pré-requisitos:** Lojinha do E37 com `src/servicos/checkout.js` e `checkout.test.js`; `ErroDeNegocio` e `FalhaNoCheckoutError` em `src/modelos/erros.js`; `useDebounce` em `src/hooks/` (E35); `EstadoCarregando`, `EstadoErro` e `EstadoVazio` no kit (E19); `.env.example` do E24.

**"async" e "await" (8 min).** No REPL do Node (`node`), com a turma:

:::steps

`src/api/cliente.js`:

```javascript
import { ApiError, ErroDeNegocio } from '../modelos/erros';

const URL_BASE = process.env.EXPO_PUBLIC_API_URL;

export async function pedir(caminho, opcoes = {}) {
  let resposta;
  try {
    resposta = await fetch(URL_BASE + caminho, {
      ...opcoes,
      headers: { 'Content-Type': 'application/json', ...opcoes.headers },
    });
  } catch (erro) {
    if (erro.name === 'AbortError') throw erro; // cancelado de propósito
    throw new ErroDeNegocio('Sem conexão. Confira a internet e tente de novo.', 'SEM_CONEXAO');
  }
  if (!resposta.ok) throw new ApiError(resposta.status);
  return resposta.json();
}
```

`src/api/produtos.js`:

```javascript
import { pedir } from './cliente';

export function traduzirProduto(p) {
  return {
    id: p.id,
    nome: p.title,
    preco: p.price, // em dólar: a API é de testes
    categoria: p.category,
    imagem: p.thumbnail,
    avaliacao: p.rating,
    descricao: p.description,
    destaque: p.rating >= 4.5,
  };
}

export async function listarProdutos({ limite = 20, pular = 0, sinal } = {}) {
  const dados = await pedir(`/products?limit=${limite}&skip=${pular}`, { signal: sinal });
  return dados.products.map(traduzirProduto);
}

export async function buscarProduto(id) {
  return traduzirProduto(await pedir('/products/' + id));
}

export async function pesquisar(termo, sinal) {
  const caminho = '/products/search?q=' + encodeURIComponent(termo.trim());
  const dados = await pedir(caminho, { signal: sinal });
  return dados.products.map(traduzirProduto);
}

export async function listarCategorias() {
  const dados = await pedir('/products/categories'); // [{ slug, name, url }]
  return dados.map((c) => ({ slug: c.slug, nome: c.name }));
}
```

```jsx
import { listarProdutos, pesquisar } from '../api/produtos';
import { Campo, EstadoCarregando, EstadoErro, EstadoVazio } from '../components/ui';
import { useDebounce } from '../hooks/useDebounce';
// saem: o import de '../data/produtos' e o de buscarProdutos ('../regras/carrinho', E27)

function buscarVitrine(termo, sinal) {
  return termo.trim() ? pesquisar(termo, sinal) : listarProdutos({ sinal });
}

export default function Inicio() {
  // useTema, useFiltros, colunas e os outros hooks continuam aqui, no topo
  const [termo, setTermo] = useState('');
  const termoAtrasado = useDebounce(termo, 300);
  const [produtos, setProdutos] = useState([]);
  const [carregando, setCarregando] = useState(true);
  const [erro, setErro] = useState(null);
  const [tentativa, setTentativa] = useState(0);

  useEffect(() => {
    const controle = new AbortController();
    async function carregar() {
      setCarregando(true);
      setErro(null);
      try {
        setProdutos(await buscarVitrine(termoAtrasado, controle.signal));
      } catch (e) {
        if (e.name !== 'AbortError') setErro(e.message);
      } finally {
        if (!controle.signal.aborted) setCarregando(false);
      }
    }
    carregar();
    return () => controle.abort();
  }, [termoAtrasado, tentativa]);

  // no E35: buscarProdutos(aplicarFiltros(produtos, filtros), termoAtrasado); a busca agora é da API
  const vitrine = aplicarFiltros(produtos, filtros);
  const esperando = termo !== termoAtrasado; // o "Buscando..." do E35 continua
```

`EntradaAnimada`, no mesmo arquivo. Com a lista vinda da API, o `entradas` e o `stagger` do E19 saem do `Inicio`, e cada cartão anima a si mesmo:

```jsx
function EntradaAnimada({ indice, style, children }) {
  const valor = useRef(new Animated.Value(0)).current;
  useEffect(() => {
    Animated.timing(valor, {
      toValue: 1, duration: 300, delay: Math.min(indice, 8) * 80, useNativeDriver: true,
    }).start();
  }, [indice, valor]);
  const subida = valor.interpolate({ inputRange: [0, 1], outputRange: [24, 0] });
  return (
    <Animated.View style={[style, { opacity: valor, transform: [{ translateY: subida }] }]}>
      {children}
    </Animated.View>
  );
}
```

## Dificuldades frequentes

| Sintoma | Causa provável | Como ajudar |
|---|---|---|
| "Sem conexão" com a internet funcionando | `.env` ausente, sem o prefixo `EXPO_PUBLIC_`, lido por desestruturação ou sem reiniciar: o endereço vira `undefined/products` | `console.log(process.env.EXPO_PUBLIC_API_URL)`; `npx expo start -c` |
| `Cannot read property 'map' of undefined` | `resposta.json()` sem `await`, ou `.products` lido numa rota de um produto só | Conferir os dois `await`; abrir a URL no navegador para ver a forma da resposta |
| Tela vermelha `Cannot read property 'interpolate' of undefined` | `entradas` do E19 criado com a lista vazia | `EntradaAnimada` com o próprio `Animated.Value` |
| Teclado fecha a cada busca | `Campo` no `ListHeaderComponent` ou no bloco que vira `EstadoCarregando` | Campo fora de `desenharConteudo` |
| "Algo deu errado" pisca ao digitar rápido | `AbortError` tratado como falha | `if (e.name !== 'AbortError')` no `catch` |
| Busca sempre vazia | Termo em português numa API em inglês | Testar "phone", "laptop" ou "watch" |
| "A loja não respondeu" em várias mesas juntas | Status 429: cerca de 100 pedidos a cada 10 segundos por IP, e a sala sai por um IP só | Esperar 10 segundos; manter o debounce; não recarregar todos juntos |
| ESLint: `React Hook useEffect has a missing dependency` | Função usada no efeito declarada fora dele | Declarar o `carregar` dentro do efeito |
