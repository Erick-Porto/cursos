```javascript
// demo-erros.mjs, fora do projeto
function calcularFrete(subtotal, cep) {
  return cep.length === 8 ? 9.9 : null; // versão simplificada do E26
}

function freteDoCep(subtotal, cep) {
  const frete = calcularFrete(subtotal, cep);
  if (frete === null) throw new Error(`CEP inválido: ${cep}`);
  return frete;
}
```

```javascript
import {
  CepInvalidoError,
  CupomInvalidoError,
  ErroDeNegocio,
  EstoqueInsuficienteError,
} from '../modelos/erros';
import { subtotal } from './carrinho';
import { calcularFrete } from './frete';
import { validarCupom } from './pedido';

export function freteDoCep(subtotal, cep) {
  const frete = calcularFrete(subtotal, cep);
  if (frete === null) {
    throw new CepInvalidoError(cep);
  }
  return frete;
}

export function conferirEstoque(itens) {
  for (const item of itens) {
    if (item.qtd > item.estoque) {
      throw new EstoqueInsuficienteError(item.nome, item.estoque);
    }
  }
}
```

`src/regras/checkout.js`, cupom e pedido:

```javascript
export function descontoDoCupom(cupom, valor) {
  if (!cupom) return 0; // cupom é opcional
  const resultado = validarCupom(cupom, valor);
  if (!resultado.valido) throw new CupomInvalidoError(cupom, resultado.mensagem);
  return resultado.desconto;
}

export function fecharPedido({ itens, cep, cupom, pagamento }) {
  if (itens.length === 0) {
    throw new ErroDeNegocio('Seu carrinho está vazio.', 'CARRINHO_VAZIO');
  }
  conferirEstoque(itens);
  const valor = subtotal(itens);
  const desconto = descontoDoCupom(cupom, valor);
  const frete = freteDoCep(valor, cep);
  return new Pedido({ itens, entrega: { cep, frete }, desconto, pagamento });
}
```

`src/contexts/PedidosContext.js`:

```jsx
import { createContext, useContext, useState } from 'react';

const PedidosContext = createContext(null);

export function PedidosProvider({ children }) {
  const [pedidos, setPedidos] = useState([]);

  // recebe o Pedido (a classe) e guarda dados simples, o mais recente primeiro
  function registrar(pedido) {
    const registro = {
      ...pedido.paraJSON(),
      id: pedidos.length + 1,
      data: new Date().toISOString(),
      total: pedido.total,
      forma: pedido.pagamento.descricao,
    };
    setPedidos([registro, ...pedidos]);
  }

  const valor = { pedidos, registrar };
  return <PedidosContext.Provider value={valor}>{children}</PedidosContext.Provider>;
}

export function usePedidos() {
  const valor = useContext(PedidosContext);
  if (!valor) throw new Error('usePedidos fora do PedidosProvider');
  return valor;
}
```

```jsx
  const [cep, setCep] = useState(params.cep ?? ''); // vem do Carrinho, agora editável
  const [codigo, setCodigo] = useState('');
  const [cupomAplicado, setCupomAplicado] = useState(null); // { codigo, desconto }

  const [parcelas, setParcelas] = useState(1);

  const [enviando, setEnviando] = useState(false);

  const frete = calcularFrete(subtotal, cep);
  const entrega = frete === null ? null : { cep, frete };
  const formas = [new PagamentoPix(), new PagamentoCartao(parcelas), new PagamentoBoleto()];
  const pagamento = formas.find((forma) => forma.tipo === tipo);
```

```jsx
  function mostrarErro(erro) {
    if (!(erro instanceof ErroDeNegocio)) throw erro; // bug: não esconda
    switch (erro.codigo) {
      case 'CEP_INVALIDO': setErros({ ...erros, cep: erro.message }); break;
      case 'CUPOM_INVALIDO': setErros({ ...erros, cupom: erro.message }); break;
      default: setAviso(erro.message); // estoque e carrinho vazio
    }
  }

  function aoSairDoCep() {
    try {
      freteDoCep(subtotal, cep);
      setErros({ ...erros, cep: null });
    } catch (erro) {
      mostrarErro(erro);
    }
  }

  function aplicarCupom() {
    try {
      const desconto = descontoDoCupom(codigo, subtotal);
      setCupomAplicado(desconto > 0 ? { codigo, desconto } : null);
      setErros({ ...erros, cupom: null });
    } catch (erro) {
      setCupomAplicado(null);
      mostrarErro(erro);
    }
  }
```

## Dificuldades frequentes

| Sintoma | Causa provável | Como ajudar |
|---|---|---|
| O teste com `toThrow` falha com o próprio erro | `expect(freteDoCep(...))` sem arrow | `expect(() => freteDoCep(...))` |
| `ReferenceError: Must call super constructor` ao criar o erro | `this.name` antes do `super(mensagem)` | `super` primeiro, como no E31 |
| Tela vermelha "só 2 em estoque" mesmo com `try` | `setItens((atual) => ...)` no contexto, ou + sem `aumentar` | `setItens(alterarQuantidade(itens, id, qtd))`, como no E28 |
| A guarda de estoque nunca dispara | Falta `estoque` no `catalogo.json` | `console.log(item)`: `qtd > undefined` é sempre falso |
| Todo erro vira "Algo deu errado" ou some | `catch` sem `instanceof`, ou `catch` vazio | Relançar o que não é `ErroDeNegocio`; mostrar `erro.message` |
| O carrinho esvazia mesmo com o CEP errado | `limpar()` antes de `fecharPedido` | Regra primeiro; registrar, navegar e limpar depois |
| Confirmação diz "Nenhum pedido por aqui" | `registrar` comentado, ou provedor dentro de uma tela | Um provedor só, no `app/_layout.js` |
| `instanceof ErroDeNegocio` falso para erro próprio | Classe sem `extends ErroDeNegocio` | Conferir o `extends` em `erros.js` |
